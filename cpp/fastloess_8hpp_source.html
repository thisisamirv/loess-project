50 Expected(T val) : val_(std::move(val)), has_val_(true) {}

55 return Expected(std::move(msg), ErrorTag{});
56 }
57
58 bool has_value() const { return has_val_; }
59
60 explicit operator bool() const { return has_val_; }
61
62 T &value() & {
63 if (!has_val_) {
64 throw LoessError(err_);
65 }
66 return val_;
67 }
68
69 const T &value() const & {
70 if (!has_val_) {
71 throw LoessError(err_);
72 }
73 return val_;
74 }
75
76 T &&value() && {

78 throw LoessError(err_);
79 }
80 return std::move(val_);
81 }
82
83 const std::string &error() const {
84 if (has_val_) {
85 throw LoessError("Bad expected access: has value");

87 return err_;
88 }

551 const std::vector<double> &y_values) {
552 if (expect_finalized_) {

554 }

557 return Expected<LoessResult>::make_error("x and y length mismatch");
558 }

561 ptr_, x_values.data(), static_cast<unsigned long>(x_values.size()),
562 y_values.data(), static_cast<unsigned long>(y_values.size()));
563
564 if (result.error != nullptr) {
565 const std::string error_msg(result.error);
566 cpp_loess_free_result(&result);
567 return Expected<LoessResult>::make_error(error_msg);
568 }
569 return Expected<LoessResult>(LoessResult(result));
570 }

573 if (expect_finalized_) {
574 return Expected<LoessResult>::make_error("Model already finalized");
575 }
576 expect_finalized_ = true;
577

579 if (result.error != nullptr) {
580 const std::string error_msg(result.error);
581 cpp_loess_free_result(&result);
