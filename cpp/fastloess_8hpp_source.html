50template <typename T> class Expected {
51public:
52 // Success constructor
53 explicit Expected(T val) : val_(std::move(val)), has_val_(true) {}
54
55 // Error constructor
56 struct ErrorTag {};
57 static Expected make_error(std::string msg) {
58 return Expected(std::move(msg), ErrorTag{});
59 }
60
61 bool has_value() const { return has_val_; }
62
63 explicit operator bool() const { return has_val_; }
64
65 T &value() & {
66 if (!has_val_) {
67 throw LoessError(err_);
68 }
69 return val_;
70 }
71
72 const T &value() const & {
73 if (!has_val_) {
74 throw LoessError(err_);
75 }
76 return val_;
77 }
78
79 T &&value() && {
80 if (!has_val_) {
81 throw LoessError(err_);
82 }
83 return std::move(val_);
84 }
85
86 const std::string &error() const {
87 if (has_val_) {
88 throw LoessError("Bad expected access: has value");
89 }
90 return err_;
91 }
92
93private:
94 Expected(std::string err, ErrorTag error_tag)
95 : err_(std::move(err)), has_val_(false) {
96 static_cast<void>(error_tag);
97 }
98
99 // We store both to avoid manual union management, relying on T's cheap
100 // default ctor. LoessResult's default ctor is cheap (zero-init).
101 T val_;
