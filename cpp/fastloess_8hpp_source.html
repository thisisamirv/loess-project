75inline void validateOutputs(const std::vector<std::string> &outputs,
76 std::initializer_list<const char *> allowed) {
77 for (const auto &output : outputs) {

82 throw LoessError("Unknown or unsupported output: " + output);

778 Expected<LoessResult> fit(const std::vector<double> &x_values,
